                    continue;
                }

                    document.cookie = cookieBase;

                    for (var clearDomainIndex = 0; clearDomainIndex < domains.length; clearDomainIndex++) {
                        var cookieDomain = domains[clearDomainIndex];

                        if (cookieDomain) {
                            document.cookie = cookieBase + ' domain=' + cookieDomain + ';';
                        }
                    }
                }
            }
