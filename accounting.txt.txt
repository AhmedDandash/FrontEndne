        window.Musa3edClearSiteCookies = function () {
            if (!document.cookie) {
                return;
            }

            var cookies = document.cookie.split(';');
            var hostName = window.location.hostname || '';
            var domainParts = hostName ? hostName.split('.') : [];
            var domains = [''];
            var paths = ['/'];
            var pathParts = (window.location.pathname || '').split('/');
            var currentPath = '';

            if (hostName) {
                domains.push(hostName);

                for (var domainIndex = 0; domainIndex < domainParts.length - 1; domainIndex++) {
                    var domainName = '.' + domainParts.slice(domainIndex).join('.');

                    if (domains.indexOf(domainName) === -1) {
                        domains.push(domainName);
                    }
                }
            }

            for (var pathIndex = 0; pathIndex < pathParts.length; pathIndex++) {
                var pathPart = pathParts[pathIndex];

                if (!pathPart) {
                    continue;
                }

                currentPath += '/' + pathPart;

                if (paths.indexOf(currentPath) === -1) {
                    paths.push(currentPath);
                }
            }

            for (var cookieIndex = 0; cookieIndex < cookies.length; cookieIndex++) {
                var cookie = cookies[cookieIndex];
                var separatorIndex = cookie.indexOf('=');
                var cookieName = (separatorIndex > -1 ? cookie.substring(0, separatorIndex) : cookie).replace(/^\s+|\s+$/g, '');

                if (!cookieName) {
                    continue;
                }

                if (cookieName.indexOf('__RequestVerificationToken') === 0) {
                    continue;
                }

                for (var clearPathIndex = 0; clearPathIndex < paths.length; clearPathIndex++) {
                    var cookiePath = paths[clearPathIndex];
                    var cookieBase = cookieName + '=; expires=Thu, 01 Jan 1970 00:00:00 GMT; Max-Age=0; path=' + cookiePath + ';';

                    document.cookie = cookieBase;

                    for (var clearDomainIndex = 0; clearDomainIndex < domains.length; clearDomainIndex++) {
                        var cookieDomain = domains[clearDomainIndex];

                        if (cookieDomain) {
                            document.cookie = cookieBase + ' domain=' + cookieDomain + ';';
                        }
                    }
                }
            }
        };

            return false;
        };
